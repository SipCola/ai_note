$projectName = Read-Host "请输入项目名（直接回车使用默认名 backend）"; if ([string]::IsNullOrWhiteSpace($projectName)) { $projectName = "backend" }; if (-not (Get-Command uv -ErrorAction SilentlyContinue)) { powershell -ExecutionPolicy ByPass -c "irm https://astral.sh/uv/install.ps1 | iex"; $env:Path = "$HOME\.local\bin;$env:Path" }; uv --version; uv init $projectName --python 3.10; cd $projectName; uv add fastapi "uvicorn[standard]" sqlmodel httpx; @'
from fastapi import FastAPI

app = FastAPI()

@app.get("/")
def hello():
    return {"message": "Hello, FastAPI!"}

@app.get("/hello/{name}")
def hello_name(name: str):
    return {"message": f"Hello, {name}!"}
